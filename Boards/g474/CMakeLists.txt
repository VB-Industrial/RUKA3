cmake_minimum_required(VERSION 3.22)
include(ExternalProject)

get_filename_component(DEFAULT_FIRMWARE_PROJECT_NAME "${CMAKE_CURRENT_SOURCE_DIR}" NAME)

set(FIRMWARE_PROJECT_NAME "${DEFAULT_FIRMWARE_PROJECT_NAME}" CACHE STRING "Firmware project name")
set(FIRMWARE_TOOLCHAIN_FILE "cmake/gcc-arm-none-eabi.cmake" CACHE STRING "Relative path to the toolchain file")
set(FIRMWARE_STM32CUBEMX_DIR "cmake/stm32cubemx" CACHE STRING "Relative path to the STM32CubeMX CMake directory")
set(FIRMWARE_APP_DIR "App" CACHE STRING "Relative path to the application sources")
set(FIRMWARE_LIBCXXCANARD_DIR "Drivers/libcxxcanard" CACHE STRING "Relative path to libcxxcanard sources")
set(FIRMWARE_LIBVOLTBRO_DIR "Drivers/libvoltbro" CACHE STRING "Relative path to libvoltbro sources")
set(FIRMWARE_CMSIS_DSP_INCLUDE_DIR "Drivers/CMSIS/DSP/Include" CACHE STRING "Relative path to CMSIS DSP headers")
set(FIRMWARE_CMSIS_DSP_LIBRARY_DIR "Drivers/CMSIS/Lib/GCC" CACHE STRING "Relative path to CMSIS DSP libraries")
set(DEFAULT_FIRMWARE_COMPILE_DEFINITIONS "ARM_MATH_CM4;STM32G4;STM32_G;MONITOR")
set(FIRMWARE_COMPILE_DEFINITIONS "${DEFAULT_FIRMWARE_COMPILE_DEFINITIONS}" CACHE STRING "Project compile definitions")
set(RUKA3_JOINT_INDEX "1" CACHE STRING "RUKA3 joint index (1..4 for STM32G474)")
option(RUKA3_BOOTSTRAP "Build the safe CAN-only commissioning application" ON)

if(NOT RUKA3_JOINT_INDEX MATCHES "^[1-4]$")
    message(FATAL_ERROR "STM32G474 target supports RUKA3_JOINT_INDEX 1..4")
endif()

if(FIRMWARE_COMPILE_DEFINITIONS STREQUAL "ARM_MATH_CM4;__FPU_PRESENT;MONITOR"
   OR FIRMWARE_COMPILE_DEFINITIONS STREQUAL "ARM_MATH_CM4;STM32G4;MONITOR")
    set(FIRMWARE_COMPILE_DEFINITIONS "${DEFAULT_FIRMWARE_COMPILE_DEFINITIONS}" CACHE STRING "Project compile definitions" FORCE)
endif()

get_filename_component(DEFAULT_STM32CUBEMX_TARGET_NAME "${FIRMWARE_STM32CUBEMX_DIR}" NAME)
set(STM32CUBEMX_TARGET_NAME "${DEFAULT_STM32CUBEMX_TARGET_NAME}" CACHE STRING "STM32CubeMX interface target name")

if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_LIST_DIR}/${FIRMWARE_TOOLCHAIN_FILE}" CACHE FILEPATH "Toolchain file")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS TRUE)

if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
    set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS Debug Release RelWithDebInfo MinSizeRel)
endif()

project(${FIRMWARE_PROJECT_NAME} LANGUAGES C CXX ASM)
message(STATUS "Build type: ${CMAKE_BUILD_TYPE}")

add_executable(${PROJECT_NAME})

if(RUKA3_BOOTSTRAP)
    set(USER_SOURCES "${FIRMWARE_APP_DIR}/bootstrap.cpp")
else()
    file(GLOB_RECURSE USER_SOURCES CONFIGURE_DEPENDS
        "${FIRMWARE_APP_DIR}/*.cpp"
    )
    list(FILTER USER_SOURCES EXCLUDE REGEX "/bootstrap\\.cpp$")
endif()
file(GLOB_RECURSE LIBCXXCANARD_SOURCES CONFIGURE_DEPENDS
    "${FIRMWARE_LIBCXXCANARD_DIR}/cyphal/*.cpp"
    "${FIRMWARE_LIBCXXCANARD_DIR}/libs/*.c"
)
file(GLOB_RECURSE LIBVOLTBRO_SOURCES CONFIGURE_DEPENDS
    "${FIRMWARE_LIBVOLTBRO_DIR}/*.cpp"
    "${FIRMWARE_LIBVOLTBRO_DIR}/*.c"
)

if(NOT DEFINED FIRMWARE_CMSIS_DSP_LIBRARY_FILE)
    set(FIRMWARE_CMSIS_DSP_LIBRARY_FILE "libarm_cortexM4lf_math.a" CACHE STRING "CMSIS DSP static library file name")
endif()

set(FIRMWARE_CMSIS_DSP_LIBRARY_PATH "${CMAKE_CURRENT_LIST_DIR}/${FIRMWARE_CMSIS_DSP_LIBRARY_DIR}/${FIRMWARE_CMSIS_DSP_LIBRARY_FILE}")
if(NOT EXISTS "${FIRMWARE_CMSIS_DSP_LIBRARY_PATH}")
    message(FATAL_ERROR "CMSIS DSP library not found: ${FIRMWARE_CMSIS_DSP_LIBRARY_PATH}")
endif()

add_library(cmsis_dsp_math STATIC IMPORTED GLOBAL)
set_target_properties(cmsis_dsp_math PROPERTIES
    IMPORTED_LOCATION "${FIRMWARE_CMSIS_DSP_LIBRARY_PATH}"
)

add_subdirectory("${FIRMWARE_STM32CUBEMX_DIR}")

target_sources(${PROJECT_NAME} PRIVATE
    ${USER_SOURCES}
    ${LIBVOLTBRO_SOURCES}
    ${LIBCXXCANARD_SOURCES}
)
if(NOT RUKA3_BOOTSTRAP)
    target_sources(${PROJECT_NAME} PRIVATE
        "${CMAKE_CURRENT_LIST_DIR}/../../App/encoder_calibration.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/../../App/calibration_service.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/../../App/joint_parameters.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/../../App/encoder_calibration_storage.c")
endif()

target_include_directories(${PROJECT_NAME} PRIVATE
    "${FIRMWARE_APP_DIR}"
    "${CMAKE_CURRENT_LIST_DIR}/../../App"
    "${FIRMWARE_LIBCXXCANARD_DIR}"
    "${FIRMWARE_LIBCXXCANARD_DIR}/libs"
    "${FIRMWARE_LIBCXXCANARD_DIR}/src"
    "${FIRMWARE_LIBVOLTBRO_DIR}"
    "${FIRMWARE_CMSIS_DSP_INCLUDE_DIR}"
)

target_compile_definitions(${PROJECT_NAME} PRIVATE
    ${FIRMWARE_COMPILE_DEFINITIONS}
    JOINT_N=${RUKA3_JOINT_INDEX}
    RUKA3_JOINT_INDEX=${RUKA3_JOINT_INDEX}
    $<$<BOOL:${RUKA3_BOOTSTRAP}>:RUKA3_BOOTSTRAP=1>
)

target_link_libraries(${PROJECT_NAME} PRIVATE
    ${STM32CUBEMX_TARGET_NAME}
    cmsis_dsp_math
)

set(RUKA3_APP_HEX "${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.hex")
set(RUKA3_FULL_HEX "${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}_full.hex")
add_custom_command(
    OUTPUT ${RUKA3_APP_HEX}
    COMMAND ${CMAKE_OBJCOPY} -O ihex $<TARGET_FILE:${PROJECT_NAME}> ${RUKA3_APP_HEX}
    DEPENDS ${PROJECT_NAME}
    VERBATIM
)
add_custom_target(${PROJECT_NAME}_hex DEPENDS ${RUKA3_APP_HEX})

set(VBBOOT_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../ThirdParty/VBBoot")
set(VBBOOT_BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}/VBBoot")
set(VBBOOT_HEX "${VBBOOT_BINARY_DIR}/VBBoot.hex")
math(EXPR RUKA3_NODE_ID "100 + ${RUKA3_JOINT_INDEX}")
ExternalProject_Add(g474_vbboot
    SOURCE_DIR ${VBBOOT_SOURCE_DIR}
    BINARY_DIR ${VBBOOT_BINARY_DIR}
    CMAKE_GENERATOR "${CMAKE_GENERATOR}"
    CMAKE_ARGS
        -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
        -DCMAKE_TOOLCHAIN_FILE=${VBBOOT_SOURCE_DIR}/cmake/gcc-arm-none-eabi.cmake
        -DCMAKE_PROJECT_VBBoot_INCLUDE=${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/vbboot_can_profile.cmake
        -DBOOTLOADER_MCU=STM32G474xx
        -DBOOTLOADER_STARTUP_FILE=${CMAKE_CURRENT_LIST_DIR}/startup_stm32g474xx.s
        -DBOOTLOADER_APP_END=0x08080000UL
        -DBOOTLOADER_RAM_END=0x20020000UL
        -DBOOTLOADER_FDCAN_CLOCK_SOURCE=PLL
        -DBOOTLOADER_FDCAN_ID_FORMAT=extended
        -DBOOTLOADER_DEFAULT_NODE_ID=${RUKA3_NODE_ID}
        -DBOOTLOADER_USE_EEPROM_CONFIG=OFF
    BUILD_COMMAND ${CMAKE_COMMAND} --build <BINARY_DIR> --clean-first
    INSTALL_COMMAND ""
    BUILD_ALWAYS TRUE
    BUILD_BYPRODUCTS ${VBBOOT_HEX}
)
add_custom_command(
    OUTPUT ${RUKA3_FULL_HEX}
    COMMAND ${CMAKE_COMMAND}
        -DBOOT_HEX=${VBBOOT_HEX}
        -DAPP_HEX=${RUKA3_APP_HEX}
        -DOUT_HEX=${RUKA3_FULL_HEX}
        -DBOOT_START=0x08000000
        -DBOOT_END=0x08003000
        -DAPP_START=0x08003000
        -DAPP_END=0x08080000
        -P ${CMAKE_CURRENT_LIST_DIR}/../g431/cmake/merge_vbdrive_hex.cmake
    DEPENDS ${PROJECT_NAME}_hex g474_vbboot ${RUKA3_APP_HEX} ${VBBOOT_HEX}
    VERBATIM
)
add_custom_target(${PROJECT_NAME}_full_hex ALL DEPENDS ${RUKA3_FULL_HEX})
